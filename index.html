SQL — Todas as tabelas
create table if not exists contratos (
  id uuid default gen_random_uuid() primary key,
  empresa text not null, tipo text not null, valor text,
  data_inicio date, data_vencimento date not null,
  responsavel text, empresa_grupo text,
  status_aprovacao text default 'Rascunho',
  aprovado_por text, data_aprovacao date,
  status_contrato text default 'Enviado',
  observacoes text, created_at timestamptz default now()
);
create table if not exists versoes (
  id uuid default gen_random_uuid() primary key,
  contrato_id uuid references contratos(id) on delete cascade,
  versao text, nome_arquivo text, url_arquivo text,
  tamanho text, criado_por text, created_at timestamptz default now()
);
create table if not exists tarefas (
  id uuid default gen_random_uuid() primary key,
  empresa text, titulo text not null, descricao text,
  responsavel text, data_limite date,
  status text default 'Pendente', created_at timestamptz default now()
);
create table if not exists demandas_externas (
  id uuid default gen_random_uuid() primary key,
  empresa text, tipo text, titulo text not null,
  descricao text, solicitante text, responsavel text,
  data_demanda date, status text default 'Concluída',
  created_at timestamptz default now()
);
create table if not exists eventos_juridicos (
  id uuid default gen_random_uuid() primary key,
  empresa text not null, tipo text, titulo text not null,
  descricao text, status text default 'Aberto',
  responsavel text, data_evento date,
  created_at timestamptz default now()
);
alter table contratos enable row level security;
alter table versoes enable row level security;
alter table tarefas enable row level security;
alter table demandas_externas enable row level security;
alter table eventos_juridicos enable row level security;
create policy "open" on contratos for all using (true);
create policy "open" on versoes for all using (true);
create policy "open" on tarefas for all using (true);
create policy "open" on demandas_externas for all using (true);
create policy "open" on eventos_juridicos for all using (true);
